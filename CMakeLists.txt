cmake_minimum_required(VERSION 3.24)
project(obs_playlist_nexus VERSION 1.0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NEXUS_BUILD_PLUGIN "Build OBS source and FFmpeg decoder" OFF)
option(NEXUS_BUILD_FRONTEND_TEST "Build isolated frontend test module (never distribute)" OFF)
enable_testing()
add_executable(fixture_generator tests/fixture_generator.cpp)
add_executable(continuity_probe_tests tests/continuity_probe_tests.cpp)
add_test(NAME continuity_detector COMMAND continuity_probe_tests)
add_executable(scheduler_tests tests/scheduler_tests.cpp)
target_include_directories(scheduler_tests PRIVATE src)
add_test(NAME scheduler COMMAND scheduler_tests)
add_executable(balanced_order_tests tests/balanced_order_tests.cpp)
target_include_directories(balanced_order_tests PRIVATE src)
add_test(NAME balanced_order COMMAND balanced_order_tests)
if(NEXUS_BUILD_PLUGIN)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 required")
  endif()
  find_path(OBS_INCLUDE_DIR obs-module.h REQUIRED)
  find_path(OBS_CONFIG_INCLUDE_DIR obsconfig.h HINTS ${OBS_INCLUDE_DIR} REQUIRED)
  find_library(OBS_LIBRARY NAMES obs libobs REQUIRED)
  find_path(FFMPEG_INCLUDE_DIR libavcodec/avcodec.h REQUIRED)
  foreach(component avcodec avformat avutil swscale)
    find_library(${component}_LIBRARY NAMES ${component} REQUIRED)
    list(APPEND FFMPEG_LIBRARIES ${${component}_LIBRARY})
  endforeach()
  add_library(obs-playlist-nexus MODULE src/plugin.cpp src/decoder.cpp src/manager.cpp)
  target_include_directories(obs-playlist-nexus PRIVATE ${OBS_INCLUDE_DIR} ${OBS_CONFIG_INCLUDE_DIR} ${FFMPEG_INCLUDE_DIR})
  target_link_libraries(obs-playlist-nexus PRIVATE ${OBS_LIBRARY} ${FFMPEG_LIBRARIES} Qt6::Widgets)
  if(MSVC)
    target_compile_options(obs-playlist-nexus PRIVATE /utf-8 /W4)
  endif()
  target_compile_definitions(obs-playlist-nexus PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_executable(decoder_tests tests/decoder_tests.cpp src/decoder.cpp)
  target_include_directories(decoder_tests PRIVATE src ${FFMPEG_INCLUDE_DIR})
  target_link_libraries(decoder_tests PRIVATE ${FFMPEG_LIBRARIES})
  target_compile_definitions(decoder_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_executable(decoder_timing_tests tests/decoder_timing_tests.cpp src/decoder.cpp)
  target_include_directories(decoder_timing_tests PRIVATE src ${FFMPEG_INCLUDE_DIR})
  target_link_libraries(decoder_timing_tests PRIVATE ${FFMPEG_LIBRARIES})
  target_compile_definitions(decoder_timing_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_executable(obs_render_tests tests/obs_render_tests.cpp)
  target_include_directories(obs_render_tests PRIVATE ${OBS_INCLUDE_DIR} ${OBS_CONFIG_INCLUDE_DIR})
  target_link_libraries(obs_render_tests PRIVATE ${OBS_LIBRARY} psapi)
  target_compile_definitions(obs_render_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_executable(manager_integration_tests tests/manager_integration_tests.cpp)
  target_include_directories(manager_integration_tests PRIVATE src ${OBS_INCLUDE_DIR} ${OBS_CONFIG_INCLUDE_DIR})
  target_link_libraries(manager_integration_tests PRIVATE ${OBS_LIBRARY} Qt6::Widgets)
  target_compile_definitions(manager_integration_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(manager_integration_tests PRIVATE /utf-8)
  endif()
  if(NEXUS_BUILD_FRONTEND_TEST)
    find_path(OBS_FRONTEND_INCLUDE_DIR obs-frontend-api.h REQUIRED)
    find_library(OBS_FRONTEND_LIBRARY NAMES obs-frontend-api REQUIRED)
    add_library(nexus-frontend-test MODULE tests/frontend_integration_module.cpp)
    target_include_directories(nexus-frontend-test PRIVATE ${OBS_INCLUDE_DIR} ${OBS_CONFIG_INCLUDE_DIR} ${OBS_FRONTEND_INCLUDE_DIR})
    target_link_libraries(nexus-frontend-test PRIVATE ${OBS_LIBRARY} ${OBS_FRONTEND_LIBRARY} Qt6::Widgets)
    target_compile_definitions(nexus-frontend-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  endif()
  foreach(component Core Gui Widgets)
    add_custom_command(TARGET obs_render_tests POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:Qt6::${component}> $<TARGET_FILE_DIR:obs_render_tests>)
  endforeach()
  install(TARGETS obs-playlist-nexus LIBRARY DESTINATION obs-plugins/64bit RUNTIME DESTINATION obs-plugins/64bit)
endif()
